```ts twoslash
// @noErrors
import { Hash, Hex, WebAuthn } from 'ox'

const credential = await WebAuthn.createCredential({ name: 'Ox' })

const challenge = Hash.keccak256(Hex.fromString('login:nonce-1234'))

const { metadata, signature } = await WebAuthn.sign({
  credentialId: credential.id,
  challenge,
})

const valid = WebAuthn.verify({
  challenge,
  metadata,
  publicKey: credential.publicKey,
  signature,
})
// @log: true
```
